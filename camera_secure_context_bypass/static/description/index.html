Developer API: Reusable OWL Component

Easily embed CameraCapture into your custom modules and OWL dialogs: 

import { Component } from "@odoo/owl";
import { CameraCapture } from "@camera_secure_context_bypass/camera_capture/camera_capture";

export class MyCustomForm extends Component {
    static components = { CameraCapture };

    onImageCaptured(result) {
        // result = { data: base64String, mimeType: "image/jpeg", width: 1280, height: 720, mode: "native" }
        console.log("Captured via mode:", result.mode);
        this.saveImage(result.data);
    }
}
